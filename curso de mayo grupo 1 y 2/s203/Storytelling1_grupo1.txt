folio + equipo + plazos + precio + anticipo + interes + subtotal + total + semanal
	+ fechaPrimerPago* + fechaUltimoPago* + fechaFinalPago*
	+ montoTotalRecibido* + montoTotalEsperado*

| fechaPrimerPago = MIN(fechaEspera)
| fechaUltimoPago = MAX(fechaPago)
| fechaFinalPago = MAX(fechaEspera)
| montoTotalRecibido = SUM(montoRecibido)
| montoTotalEsperado = SUM(montoEsperado)

/ montoExcedenteEsperado = if montoTotalEsperado > total then montoTotalEsperado - total else 0
/ montoExcedenteRecibido = if montoTotalRecibido > total then montoExcedenteRecibido - total else 0
/ montoTotalFaltante = (montoTotalEsperado - montoExcedenteEsperado) - (montoTotalRecibido - montoExcedenteRecibido)
/ diasActivo = (fechaUltimoPago - fechaPrimerPago) + 1semana
/ diasTotales = (fechaFinalPago - fechaPrimerPago)
/ diasFaltantes = fechaFinalPago - (fechaUltimoPago + 1semana)

: Monto Recaudado Inicial = anticipo
: % Recaudado Inicial = Monto Recaudado Inicial / precio
: Monto En Riesgo = precio - anticipo
: % En Riesgo = Monto En Riesgo / precio
: % Progreso Activo = diasActivo / diasTotales
: % Progreso Faltante = diasFaltantes / diasTotales
: Monto Total Invertido = precio
: Monto Total = total
: Monto Total Esperado = montoTotalEsperado - montoExcedenteEsperado
: Monto Total Recibido = montoTotalRecibido - montoExcedenteRecibido
: Monto Total Faltante = Monto Total Esperado - Monto Total Recibido
: % Progreso Esperado = Monto Total Esperado / Monto Total
: % Progreso Recibido = Monto Total Recibido / Monto Total
: % Progreso Faltante = Monto Total Faltante / Monto Total
: % Retraso = % Progreso Esperado - % Progreso Pagado
: Margen Total = Monto Total - Monto Total Invertido
: Margen Total Esperado =  Monto Total Esperado - Monto Total Invertido
: % Margen Total = Margen Total / Monto Total Invertido
: % Margen Total Esperado = Margen Total Esperado / Monto Total Invertido
